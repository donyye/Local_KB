+在把镜像上传到我们自己的镜像仓库的时候需要先配置 daemon.json

root@k81node:~# vim /etc/docker/daemon.json

{

[    "registry-mirrors": \"[https://9v94unqh.mirror.aliyuncs.com](https://9v94unqh.mirror.aliyuncs.com)["],][   ]\<--- 记得这里加个"," 号

[    ]["insecure-registries": ["10.10.40.81:1000"]][    ]\<-- 添加这一行

}

 

+然后加载并重启服务

root@k81node:~# systemctl daemon-reload 

 

root@k81node:~# systemctl restart docker
