dev nvme10n1

dev nvme11n1

dev nvme6n1

dev nvme7n1

dev nvme8n1

/dev/nvme6n1 ; serial: 11A0A0K0TAHR

/dev/nvme7n1 ; serial: 11A0A0H7TAHR

/dev/nvme8n1 ; serial: 11A0A0JJTAHR

/dev/nvme9n1 ; serial: 11A0A0H1TAHR

/dev/nvme10n1 ; serial: 11A0A0KBTAHR

/dev/nvme11n1 ; serial: 11A0A0JBTAHR
