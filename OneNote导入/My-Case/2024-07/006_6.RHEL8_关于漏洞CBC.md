自建配置文件：

[[root@rh89 ~]#] cat /etc/crypto-policies/policies/modules/DISABLE-CBC.pmod

tls_cipher = -AES-256-CBC -AES-128-CBC

cipher = -AES-128-CBC -AES-256-CBC -CAMELLIA-256-CBC -CAMELLIA-128-CBC

ssh_cipher = -AES-128-CBC -AES-256-CBC
