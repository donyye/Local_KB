3. 配置bond0

\# nmcli con add type bond con-name bond0 ifname bond0 mode active-backup

 

\# nmcli con add type bond-slave ifname ens35 master bond0

 

\# nmcli con add type bond-slave ifname ens36 master bond0

 

[\[root@rh8kvm1 network-scripts\]#] ls

ifcfg-bond0  ifcfg-bond-slave-ens35  ifcfg-bond-slave-ens36

==== 做桥接 ====

\[root@rh8kvm1 network-scripts\]# nmcli con add type bridge con-name br0 ifname br0

 

\[root@rh8kvm1 network-scripts\]# ls

ifcfg-br0[  ......]

 

\[root@rh8kvm1 network-scripts\]# nmcli con mod br0 ipv4.addresses \"10.10.41.133/16\" ipv4.method manual autoconnect yes

 

\[root@rh8kvm1 network-scripts\]# cat ifcfg-br0

STP=yes

BRIDGING_OPTS=priority=32768

TYPE=Bridge

PROXY_METHOD=none

BROWSER_ONLY=no

BOOTPROTO=none

DEFROUTE=yes

IPV4_FAILURE_FATAL=no

IPV6INIT=yes

IPV6_AUTOCONF=yes

IPV6_DEFROUTE=yes

IPV6_FAILURE_FATAL=no

IPV6_ADDR_GEN_MODE=stable-privacy

NAME=br0

UUID=fa7eea9b-6cf7-4448-87cb-16f1739e8fd5

DEVICE=br0

ONBOOT=yes

IPADDR=10.10.41.133

PREFIX=16

           

\[root@rh8kvm1 \~\]# nmcli con mod bond0 master br0

 

\[root@rh8kvm1 network-scripts\]# cat ifcfg-bond0

BONDING_OPTS=mode=active-backup

TYPE=Bond

BONDING_MASTER=yes

NAME=bond0

UUID=32ab7416-ac10-42ef-818a-77684ff4250e

DEVICE=bond0

ONBOOT=yes

BRIDGE=br0[   ][ \<\-\-- ]多了个设置，少了很多其它的配置。

 

启动br0：

\[root@rh8kvm1 network-scripts\]# nmcli con up br0

Connection successfully activated (master waiting for slaves) (D-Bus active path: /org/freedesktop/NetworkManager/ActiveConnection/16)
