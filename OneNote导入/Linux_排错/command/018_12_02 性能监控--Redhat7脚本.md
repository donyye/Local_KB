Redhat 性能记录脚本

<https://access.redhat.com/articles/279063>

 

21 \* \* \* \* bash /path/to/watcher-cron.bsh 10 361 \<logdir\> 4 1

注意：将 /path/to 替换为您环境中脚本的路径。

注意：将\<logdir\>替换为您想要存储日志文件的目录。如 /var/mylog

建议不要将脚本和日志目录都放在任何存在可疑问题的存储上。

10 代表10秒收集一次数据。 

361 代表运行361此。

10 \* 361 / 60 = 60.xxx 大概是60分钟。

也就收集1个小时的数据，而每个小时的数据会产生一个压缩包。

如这样：

20191010-112101-info-watcher.tar.gz

20191010-122101-info-watcher.tar.gz

20191010-132101-info-watcher.tar.gz

=================

具体的脚本如下：

#\*

#\*\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\--

#\*

#\* Maintainer: bubrown@redhat.com

\#

\# watcher-cron.bsh V02.801

\#

\# \$1 = interval

\# \$2 = interations

\# \$3 = log directory

\# \$4 = days of logs to keep

\# \$5 = compression level (optional)

\# 0 - no compression (previous default)

\# 1 - end-of-batch compression (new default, about a 10:1 compression ratio observed)

\# 2 - \'1\' + use gzip, if available, inline with top output creation

\# 3 - \'2\' + use gzip all commands except vmstat

\#

\# crontab -u root -e

\# 21 \* \* \* \* bash /\<path\>/watcher-cron.bsh 30 120 /\<logdir\> 2 1

\#

if \[ \"\$4\" == \"\" \]

then

echo \" \"

echo \"missing arguments:\"

echo \"\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\--\"

echo \" arg1 = interval, e.g. 1 (in seconds)\"

echo \" arg2 = interations, e.g. 720 (count of samples to take)\"

echo \" (from cron arg1 x arg2 = 3600 seconds, 1hr)\"

echo \" arg3 = log path, e.g. /tmp/watcher (where to create log files)\"

echo \" arg4 = days, e.g. 2 (anything older than this is purged from log path)\"

echo \"\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\-\--\"

echo \" \"

exit

fi

\# -p is exclusive of -x on RHEL5\*

\# -p is required along with -x on RHEL6\* to get partitions included

\# -p is required along with -x on RHEL7\* to get partitions included

\_pflg=\"\"

if \[\[ \`uname -a\` == \*2.6.32\* \|\| \`uname -a\` == \*3.10.0\* \]\]

then

 \_pflg=\"-p\"

fi

\_compress_hourly=1

\_compress_top=\"\"

\_compress_all=\"\"

\_ext_top=\"\"

\_ext_all=\"\"

if \[ \"\$5\" != \"\" \]

then

  if \[ \$5 -eq 0 \]

  then

     \_compress_hourly=0

  fi

  if \[ -e \"/usr/bin/gzip\" \]

  then

    if \[ \$5 -ge 2 \]

    then

      \_compress_top=\" \| gzip - \"

      \_ext_top=\".gz\"

    fi

    if \[ \$5 -ge 3 \]

    then

      \_compress_all=\" \| gzip - \"

      \_ext_all=\".gz\"

     fi

  fi

fi

\_wait_til_complete=0

if \[ \$\_compress_hourly -gt 0 \]

then

   \_wait_til_complete=1

fi

\_time=\$(date +%Y%m%d-%H%M%S)

\_name=\$(uname -a)

#echo \$1 seconds x \$2 intervals, at \$

echo \'#TIME= \'\$ \> \$3/\$-info-vmstat.log

echo \'#INTERVAL= \'\$1 \>\> \$3/\$-info-vmstat.log

echo \'#COUNT= \'\$2 \>\> \$3/\$-info-vmstat.log

echo \'#UNAME= \"\'\$\'\"\' \>\> \$3/\$-info-vmstat.log

\#

echo \'#TIME= \'\$ \> \$3/\$-info-vmstat-d.log

echo \'#INTERVAL= \'\$1 \>\> \$3/\$-info-vmstat-d.log

echo \'#COUNT= \'\$2 \>\> \$3/\$-info-vmstat-d.log

echo \'#UNAME= \"\'\$\'\"\' \>\> \$3/\$-info-vmstat-d.log

\#

 eval \"echo \'#UNAME= \\\"\$\\\"\' \$ \> \$3/\$-info-iostat.log\$\"

#eval \"iostat \$1 \$2 -t -k -x \$ -n \$ \>\> \$3/\$-info-iostat.log\$ &\"

 eval \"iostat \$1 \$2 -t -k -x \$ \$ \>\> \$3/\$-info-iostat.log\$ &\" ; iopid=\$!

       vmstat -t \$1 \$2 \>\> \$3/\$-info-vmstat.log & vmpid=\$!

       vmstat -t \$1 \$2 -d \>\> \$3/\$-info-vmstat-d.log & vdpid=\$!

 eval \"top -b -d \$1 -n \$2 \$ \> \$3/\$-info-top.log\$ &\" ; tppid=\$!

 eval \"mpstat \$1 \$2 -P ALL \$ \> \$3/\$-info-mpstat.log\$ &\" ; mppid=\$!

\#

if \[ \$\_wait_til_complete -gt 0 \]

then

  sleep \$(( \$1 \* \$2 ))

fi

find \$3/\*-info-\*.log\* -mtime +\$4 -exec rm -f  \\;

if \[ \$\_compress_hourly -gt 0 \]

then

  while sleep 1; do ps -o pid= -p \$iopid \|\| break; done

  while sleep 1; do ps -o pid= -p \$vmpid \|\| break; done

  while sleep 1; do ps -o pid= -p \$vdpid \|\| break; done

  while sleep 1; do ps -o pid= -p \$tppid \|\| break; done

  while sleep 1; do ps -o pid= -p \$mppid \|\| break; done

  ls -lh \$3/\$-info\*.log\* \> \$3/\$-info-files.log

  ls -1c \$3 \| grep \$-info \| grep log \> \$3/\$-files.txt

  tar -C \$3 -T \$3/\$-files.txt \--remove-files -czf \$3/\$-info-watcher.tar.gz

  find \$3/\*-info-watcher.tar.gz -mtime +\$4 -exec rm -f  \\;

  rm -f \$3/\$-files.txt

fi
