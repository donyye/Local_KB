1. IPA-client-install 在 IPA 中创建主机条目，但不创建 DNS 记录

<https://access.redhat.com/solutions/3156971>

ipa-client-install install \--mkhomedir\
\"Failed to update DNS records\"

遇到的案例：

![[idm部署与使用_009_2.4 idm一些设置_001.png]]

 

使用以下命令修改 IPA 服务器上的 BIND 策略:

\[root@master \~\]# ipa dnszone-mod  idm.domain.com. \--dynamic-update=TRUE

  Zone name: idm.domain.com.

  Active zone: True

  Authoritative nameserver: master.idm.domain.com.

  Administrator e-mail address: hostmaster.idm.domain.com.

  SOA serial: 1686703292

  SOA refresh: 3600

  SOA retry: 900

  SOA expire: 1209600

  SOA minimum: 3600

  BIND update policy: grant IDM.DOMAIN.COM krb5-self \* A; grant IDM.DOMAIN.COM krb5-self \* AAAA; grant IDM.DOMAIN.COM krb5-self \* SSHFP;

[  Dynamic update: True]

  Allow query: any;

  Allow transfer: none;

 

启用自动 PTR 记录同步

[\[root@master \~\]# ipa dnszone-mod idm.domain.com \--allow-sync-ptr=TRUE]

  Zone name: idm.domain.com.

  Active zone: True

  Authoritative nameserver: master.idm.domain.com.

  Administrator e-mail address: hostmaster.idm.domain.com.

  SOA serial: 1686703292

  SOA refresh: 3600

  SOA retry: 900

  SOA expire: 1209600

  SOA minimum: 3600

  BIND update policy: grant IDM.DOMAIN.COM krb5-self \* A; grant IDM.DOMAIN.COM krb5-self \* AAAA; grant IDM.DOMAIN.COM krb5-self \* SSHFP;

  Dynamic update: True

  Allow query: any;

  Allow transfer: none;

[  Allow PTR sync: True]

 

 

[\[root@master \~\]# ipa dnszone-mod 40.10.10.in-addr.arpa. \--allow-sync-ptr=TRUE]

  Zone name: 40.10.10.in-addr.arpa.

  Active zone: True

  Authoritative nameserver: master.idm.domain.com.

  Administrator e-mail address: hostmaster

  SOA serial: 1686731685

  SOA refresh: 3600

  SOA retry: 900

  SOA expire: 1209600

  SOA minimum: 3600

  BIND update policy: grant IDM.DOMAIN.COM krb5-subdomain 40.10.10.in-addr.arpa. PTR;

  Dynamic update: False

  Allow query: any;

  Allow transfer: none;

[  Allow PTR sync: True]

 

 

查看 BIND 策略

\[root@master \~\]# ipa dnszone-show idm.domain.com
