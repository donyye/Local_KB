Host:

[\[root@localhost \~\]# lspci -nnD \|grep -i nvidia]

0000:65:00.0 3D controller \[0302\]: NVIDIA Corporation GA100 \[A100 PCIe 80GB\] \[10de:20b5\] (rev a1)

0000:ca:00.0 3D controller \[0302\]: NVIDIA Corporation GA100 \[A100 PCIe 80GB\] \[10de:20b5\] (rev a1)

 

cat /etc/modprobe.d/vfio.conf 

options vfio-pci ids=10de:20b5

softdep nvidia pre: vfio-pci

 

 

cat /etc/modules-load.d/vfio-pci.conf 

vfio-pci

 

cat /etc/default/grub

......

GRUB_CMDLINE_LINUX=\"crashkernel=auto resume=/dev/mapper/rhel_anchorhost-swap rd.lvm.lv=rhel_anchorhost/root rd.lvm.lv=rhel_anchorhost/swap intel_iommu=on[  rd.driver.pre=vfio-pci  pci-stub.ids=10de:10de:20b5 iommu=pt] rhgb quiet rdblacklist=nouveau\"

......

grub2-mkconfig -o /etc/grub2.cfg

就添加这些，其它不要添加
