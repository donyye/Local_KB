\-\--

\- name: install1

  hosts: dev,test,prod

  tasks:

  - name: install packages

    ansible.builtin.yum:

      name: \"{}\"

      state: present

    vars:

      packages:

      - php

      - mariadb

 

\- name: install2

  hosts: dev

  tasks:

  - name: install group

    ansible.builtin.yum:

      name: \"@RPM Development Tools\"

      state: present

  - name: upgrade all packages

    ansible.builtin.yum:

      name: \'\*\'

      state: latest
