2025年2月6日

11:06

\[greg@control ansible\]\$ vim /home/greg/ansible/webcontent.yml

\-\--

\- name: Create Web Directory

  hosts: dev

  roles:

  - apache

  tasks:

  - name: Create directory

    ansible.builtin.file:

      path: /webdev

      state: directory

      group: webdev

      mode: \'2775\'

    

  - name: Create link

    ansible.builtin.file:

[      src: /webdev     ]#\--\> 源头

[      dest: /var/www/html/webdev  ][ \# \--\> ]连接到 webdev 源头

      state: link

  

  - name: Copy content

    ansible.builtin.copy:

      content: \'Development\'

      dest: /webdev/index.html

      setype: httpd_sys_content_t

 

\[greg@control ansible\]\$ ansible-navigator run webcontent.yml -m stdout
