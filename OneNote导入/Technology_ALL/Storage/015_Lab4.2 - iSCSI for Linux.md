  实验到此结束

   

  1.  rpm -ivh iscsi.rpm
  2.  dkms.rpm
  3.  dm/install.sh
  4.  iscsiadm -m iface -I iscsi1 -o new

  iscsiadm -m iface -I iscsi2 -o new

  1.  iscsiadm -m -iface -I iscsi1 \--op=update -n iface.net_ifacename -v eth1

  iscsiadm -m -iface -I iscsi2 \--op=update -n iface.net_ifacename -v eth2

  1.  ls /var/lib/iscsi/ifaces
  2.  fdisk /dev/mapper/mpatha
  3.  partprobe
  4.  mkfs.ext3 /dev/mapper/mpathap1

  mkfs.ext3 /dev/mapper/mpathap2

  1.  vi /etc/fstab
